\newcommand{\FirstPageBackground}{
    \begin{tikzpicture}[remember picture, overlay]
        \fill[Dark_blue]
            (current page.south west) ++(0cm, 0cm) rectangle ++(22cm, 30cm);
        \fill[Light_blue]
            (current page.south west) ++(2cm, 2cm) rectangle ++(17cm, 26cm);
        \fill[Medium_blue]
            (current page.south west) ++(4cm, 4.5cm) rectangle ++(13cm, 10.3cm);
    \end{tikzpicture}
}

% Fondo para el resto de las páginas
\newcommand{\OtherPageBackground}{
    \begin{tikzpicture}[remember picture, overlay]
        \fill[Light_Gray](current page.south west) ++(0cm, 0cm) rectangle ++(5.5cm, 30cm);
    \end{tikzpicture}
}

% Configuración inicial del fondo
\backgroundsetup{scale=1, color=white, opacity=1, angle=0, contents={\FirstPageBackground}}

% --- fondo página 1 ---
\AtBeginShipoutFirst{%
  \backgroundsetup{contents={\FirstPageBackground}}%
  \BgThispage
}

% --- resto de páginas: 2 tiene fondo, >=3 sin fondo ---
\AtBeginShipout{%
  \ifnum\value{page}=1\relax
    % nada: la página 1 ya fue tratada por \AtBeginShipoutFirst
  \else
    \ifnum\value{page}=2\relax
      \backgroundsetup{contents={\OtherPageBackground}}%
      \BgThispage
    \else
      % desde la página 3 en adelante: forzamos fondo vacío
      \backgroundsetup{contents={}}%
      \BgThispage
    \fi
  \fi
}
